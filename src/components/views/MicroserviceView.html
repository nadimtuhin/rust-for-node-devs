1. Entrypoint & Graceful Shutdown with Pool Draining

In Axum 0.7+, axum::Server was replaced with tokio::net::TcpListener and axum::serve(). Graceful shutdown intercepts both SIGINT (Ctrl+C) and Unix SIGTERM (Kubernetes pod eviction). Crucially, once the HTTP server finishes in-flight requests, the application explicitly awaits db.close().await to drain all PostgreSQL pooled connections cleanly before process exit.

// src/main.rs
use axum::{routing::{get, post}, Router};
use std::sync::Arc;
use tokio::net::TcpListener;
use tower_http::trace::TraceLayer;

mod error;
mod routes;
mod state;

use state::AppState;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    // 1. Initialize structured logging
    tracing_subscriber::fmt::init();

    // 2. Build shared thread-safe application state
    let state = AppState::new().await?;

    // 3. Assemble routes with State and Tower middleware
    let app = Router::new()
        .route("/healthz", get(routes::health::check))
        .route("/api/users", post(routes::users::create_user))
        .route("/api/users/:id", get(routes::users::get_user))
        .layer(TraceLayer::new_for_http())
        .with_state(state.clone());

    // 4. Bind raw Tokio TCP listener (Axum 0.7 idiom)
    let addr = "0.0.0.0:3000";
    let listener = TcpListener::bind(addr).await?;
    tracing::info!("Listening on {}", addr);

    // 5. Serve with graceful SIGINT / SIGTERM drain
    axum::serve(listener, app)
        .with_graceful_shutdown(shutdown_signal())
        .await?;

    // 6. Explicit pool draining: close connections and wait for in-flight DB queries
    tracing::info!("Draining PostgreSQL connection pool...");
    state.db.close().await;
    tracing::info!("PostgreSQL pool closed. Server shutdown complete.");

    Ok(())
}

async fn shutdown_signal() {
    let ctrl_c = async {
        tokio::signal::ctrl_c()
            .await
            .expect("failed to install Ctrl+C handler");
    };

    #[cfg(unix)]
    let terminate = async {
        let mut sig = tokio::signal::unix::signal(tokio::signal::unix::SignalKind::terminate())
            .expect("failed to install SIGTERM handler");
        sig.recv().await;
    };

    #[cfg(not(unix))]
    let terminate = std::future::pending::<()>();

    tokio::select! {
        _ = ctrl_c => tracing::info!("Received SIGINT (Ctrl+C), draining HTTP connections..."),
        _ = terminate => tracing::info!("Received SIGTERM, draining HTTP connections..."),
    }
}

2. Dependency Injection: State, FromRef & Extractor Ordering Rules

Unlike Express's untyped req.app.locals, Axum uses type-safe extractor DI. Implement FromRef to allow handlers to extract only the sub-state they need.

Cardinal Extractor Ordering Rule:

Extractors implementing FromRequestParts (e.g. Path, Query, State, HeaderMap) inspect only request metadata and must precede body-consuming extractors.
Any extractor implementing FromRequest that consumes the HTTP request body stream (e.g. Json, String, Bytes) MUST be placed last in the argument list. Violating this triggers the compile error: the trait Handler<...> is not implemented for fn(...). 

// src/state.rs
use axum::extract::FromRef;
use sqlx::{Pool, Postgres, postgres::PgPoolOptions};

#[derive(Clone)]
pub struct AppState {
    pub db: Pool<Postgres>,
}

impl AppState {
    pub async fn new() -> Result<Self, sqlx::Error> {
        let db_url = std::env::var("DATABASE_URL").expect("DATABASE_URL must be set");
        let db = PgPoolOptions::new()
            .max_connections(20)
            .connect(&db_url)
            .await?;
        Ok(Self { db })
    }
}

// Enables handlers to extract State(db): State<Pool<Postgres>> directly
impl FromRef<AppState> for Pool<Postgres> {
    fn from_ref(app_state: &AppState) -> Self {
        app_state.db.clone()
    }
}
// src/routes/users.rs
use axum::{
    extract::{Path, Query, State},
    http::StatusCode,
    Json,
};
use serde::{Deserialize, Serialize};
use sqlx::{Pool, Postgres};
use uuid::Uuid;
use crate::error::AppError;

#[derive(Deserialize)]
pub struct CreateUserPayload {
    pub email: String,
}

#[derive(Deserialize)]
pub struct PaginationQuery {
    #[serde(default = "default_page")]
    pub page: u32,
    #[serde(default = "default_limit")]
    pub limit: u32,
}

fn default_page() -> u32 { 1 }
fn default_limit() -> u32 { 20 }

#[derive(Serialize, sqlx::FromRow)]
pub struct UserResponse {
    pub id: Uuid,
    pub email: String,
}

// ORDER RULE: State and RequestParts come first. Json (body consumer) is LAST!
pub async fn create_user(
    State(db): State<Pool<Postgres>>,
    Json(payload): Json<CreateUserPayload>, // Terminal: consumes request stream
) -> Result<(StatusCode, Json<UserResponse>), AppError> {
    let user = sqlx::query_as::<_, UserResponse>(
        "INSERT INTO users (id, email) VALUES ($1, $2) RETURNING id, email"
    )
    .bind(Uuid::new_v4())
    .bind(&payload.email)
    .fetch_one(&db)
    .await?;

    Ok((StatusCode::CREATED, Json(user)))
}

// GET /api/users/{id}?page=1&limit=20
// Demonstrates Path, Query, and State all implementing FromRequestParts
pub async fn get_user(
    Path(id): Path<Uuid>,
    Query(pagination): Query<PaginationQuery>,
    State(db): State<Pool<Postgres>>,
) -> Result<Json<UserResponse>, AppError> {
    let offset = (pagination.page.saturating_sub(1) as i64) * (pagination.limit as i64);
    let user = sqlx::query_as::<_, UserResponse>(
        "SELECT id, email FROM users WHERE id = $1 LIMIT $2 OFFSET $3"
    )
    .bind(id)
    .bind(pagination.limit as i64)
    .bind(offset)
    .fetch_optional(&db)
    .await?
    .ok_or(AppError::NotFound("User not found".into()))?;

    Ok(Json(user))
}
