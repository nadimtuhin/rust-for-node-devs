Testing & Mocking: Jest / Vitest vs Rust

First-class built-in test runner, parallel execution by default, and trait-based dependency injection.

Built-in Test Module (Unit Testing)

pub fn add(a: i32, b: i32) -> i32 {
    a + b
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_add() {
        assert_eq!(add(2, 2), 4);
    }

    #[test]
    #[should_panic(expected = "divide by zero")]
    fn test_panics() {
        panic!("divide by zero");
    }
}
