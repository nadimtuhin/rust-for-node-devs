1. High-Throughput Token Streaming Gateway (SSE)

Node.js struggles with thousands of concurrent HTTP chunked/SSE connections due to V8 heap bloat. In Rust, Axum handles 100k+ SSE streams with zero allocations per frame using tokio_stream and async channels.

use axum::{
    response::sse::{Event, KeepAlive, Sse},
    routing::post,
    Json, Router,
};
use futures::stream::{Stream, StreamExt};
use serde::{Deserialize, Serialize};
use std::{convert::Infallible, time::Duration};
use tokio::sync::mpsc;
use tokio_stream::wrappers::ReceiverStream;

#[derive(Deserialize)]
pub struct ChatRequest {
    pub prompt: String,
}

pub async fn chat_stream_handler(
    Json(req): Json<ChatRequest>,
) -> Sse<impl Stream<Item = Result<Event, Infallible>>> {
    let (tx, rx) = mpsc::channel::<String>(100);

    // Spawn inference worker task
    tokio::spawn(async move {
        let tokens = vec!["Rust", " provides", " fearless", " concurrency", " for", " AI."];
        for token in tokens {
            tokio::time::sleep(Duration::from_millis(50)).await;
            if tx.send(token.to_string()).await.is_err() {
                break; // Client disconnected early
            }
        }
    });

    // Adapt mpsc receiver to SSE Stream
    let stream = ReceiverStream::new(rx).map(|token| {
        Ok(Event::default().data(token))
    });

    Sse::new(stream).keep_alive(KeepAlive::new().interval(Duration::from_secs(15)))
}

2. Declarative Agent & Tool Execution with rig-core

rig-core provides type-safe LLM tool orchestration equivalent to LangChain or Vercel AI SDK, but without memory leaks or runtime type coercion errors.

use rig::{
    completion::Prompt,
    providers::openai::{Client, GPT_4O},
    tool::Tool,
};
use serde::{Deserialize, Serialize};

#[derive(Deserialize)]
struct CalcArgs { a: f64, b: f64 }

#[derive(Deserialize, Serialize)]
struct CalcTool;

impl Tool for CalcTool {
    const NAME: &'static str = "calculate_sum";
    type Error = std::io::Error;
    type Args = CalcArgs;
    type Output = f64;

    async fn definition(&self, _prompt: String) -> rig::tool::ToolDefinition {
        serde_json::from_value(serde_json::json!({
            "name": "calculate_sum",
            "description": "Calculates the sum of two floating point numbers",
            "parameters": {
                "type": "object",
                "properties": { "a": {"type": "number"}, "b": {"type": "number"} },
                "required": ["a", "b"]
            }
        })).unwrap()
    }

    async fn call(&self, args: Self::Args) -> Result<Self::Output, Self::Error> {
        Ok(args.a + args.b)
    }
}

pub async fn run_agent() -> Result<String, Box<dyn std::error::Error>> {
    let client = Client::from_env();
    let agent = client.agent(GPT_4O)
        .preamble("You are a financial analysis agent.")
        .tool(CalcTool)
        .build();

    let response = agent.prompt("What is 142.50 plus 391.25?").await?;
    Ok(response)
}

3. Zero-Python Local Inference with Hugging Face Candle

Run quantized LLMs directly in your microservice without spawning heavy Python or ONNX runtimes using Metal or CUDA hardware acceleration.

use candle_core::{Device, Tensor};
use candle_transformers::generation::LogitsProcessor;

pub fn sample_next_token(
    logits: &Tensor, 
    logits_processor: &mut LogitsProcessor
) -> Result<u32, candle_core::Error> {
    // Select compute target: Metal (Apple Silicon), Cuda (Nvidia), or CPU
    let device = Device::new_metal(0).unwrap_or(Device::Cpu);

    // Extract raw logits for final token position: [batch_size, seq_len, vocab_size]
    let next_token_logits = logits.squeeze(0)?.get(logits.dim(1)? - 1)?;

    // Sample next token ID using temperature & top-p sampling
    let next_token_id = logits_processor.sample(&next_token_logits)?;
    Ok(next_token_id)
}
