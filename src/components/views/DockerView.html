Appendix 05 • Containerization

Production Multi-Stage Dockerfile

Build static binaries with cargo-chef caching and Google Distroless for minimal attack surfaces.

The Gold-Standard Dockerfile

# Stage 1: Dependency Chef Recipe
FROM rust:1.80-bullseye AS chef
RUN cargo install cargo-chef
WORKDIR /app

# Stage 2: Prepare lockfile recipe
FROM chef AS planner
COPY . .
RUN cargo chef prepare --recipe-path recipe.json

# Stage 3: Build cached dependencies
FROM chef AS builder
COPY --from=planner /app/recipe.json recipe.json
RUN cargo chef cook --release --recipe-path recipe.json
COPY . .
RUN cargo build --release --bin server

# Stage 4: Ultra-minimal production runtime (< 15MB)
FROM gcr.io/distroless/cc-debian12:nonroot
WORKDIR /app
COPY --from=builder /app/target/release/server /app/server
USER nonroot:nonroot
EXPOSE 8080
ENTRYPOINT ["/app/server"]
