1. High-Throughput Token Streaming Gateway (Axum 0.7 SSE)

Node.js struggles with thousands of concurrent HTTP chunked/SSE connections due to V8 heap overhead (~35KB+ per active socket) and event loop GC pauses. In Rust, Axum handles 100k+ SSE connections using Tokio tasks with lightweight (~2KB) memory footprints and zero frame-allocations.

use axum::{
    response::sse::{Event, KeepAlive, Sse},
    routing::post,
    Json, Router,
};
use futures_util::stream::{Stream, StreamExt};
use serde::Deserialize;
use std::{convert::Infallible, time::Duration};
use tokio::sync::mpsc;
use tokio_stream::wrappers::ReceiverStream;

#[derive(Deserialize)]
pub struct ChatRequest {
    pub prompt: String,
}

pub async fn chat_stream_handler(
    Json(req): Json<ChatRequest>,
) -> Sse<impl Stream<Item = Result<Event, Infallible>>> {
    // Tight bounded channel (16 tokens) gives immediate backpressure if client lags
    let (tx, rx) = mpsc::channel::<String>(16);

    // Spawn inference worker task
    tokio::spawn(async move {
        let tokens = vec!["Rust", " provides", " fearless", " concurrency", " for", " AI."];
        for token in tokens {
            tokio::time::sleep(Duration::from_millis(50)).await;
            // Early break when HTTP connection drops
            if tx.send(token.to_string()).await.is_err() {
                break;
            }
        }
    });

    // Adapt tokio mpsc receiver into an Axum SSE stream
    let stream = ReceiverStream::new(rx).map(|token| {
        Ok(Event::default().data(token))
    });

    Sse::new(stream).keep_alive(
        KeepAlive::new()
            .interval(Duration::from_secs(15))
            .text("keep-alive-ping")
    )
}

2. Declarative Agent & Tool Execution with rig-core

rig-core is the Rust analog to LangChain.js and Vercel AI SDK. Unlike JavaScript runtime schemas (Zod) prone to unhandled JSON coercion exceptions, Rig enforces compile-time Serde typed schemas and async tool orchestration.

use rig::{
    completion::Prompt,
    providers::openai::{Client, GPT_4O},
    tool::Tool,
};
use serde::{Deserialize, Serialize};
use serde_json::json;

#[derive(Deserialize)]
struct CalcArgs { 
    a: f64, 
    b: f64 
}

#[derive(Deserialize, Serialize)]
struct CalcTool;

impl Tool for CalcTool {
    const NAME: &'static str = "calculate_sum";
    type Error = std::io::Error;
    type Args = CalcArgs;
    type Output = f64;

    fn description(&self) -> String {
        "Calculates the sum of two numbers".to_string()
    }

    fn parameters(&self) -> serde_json::Value {
        json!({
            "type": "object",
            "properties": {
                "a": { "type": "number", "description": "First operand" },
                "b": { "type": "number", "description": "Second operand" }
            },
            "required": ["a", "b"]
        })
    }

    async fn call(&self, args: Self::Args) -> Result<Self::Output, Self::Error> {
        Ok(args.a + args.b)
    }
}

pub async fn run_agent() -> Result<String, Box<dyn std::error::Error + Send + Sync>> {
    let client = Client::from_env();
    let agent = client.agent(GPT_4O)
        .preamble("You are a financial analysis assistant.")
        .tool(CalcTool)
        .build();

    let response = agent.prompt("What is 142.50 plus 391.25?").await?;
    Ok(response)
}

3. Zero-Python Local Inference with Hugging Face Candle

Run quantized GGUF/safetensors models natively in your service without Python, ONNX, or C++ FFI bindings, dispatching dynamically to Metal (Apple Silicon), CUDA, or CPU.

use candle_core::{Device, IndexOp, Tensor};
use candle_transformers::generation::LogitsProcessor;

pub fn select_device() -> Device {
    #[cfg(feature = "cuda")]
    {
        Device::cuda_if_available(0).unwrap_or(Device::Cpu)
    }
    #[cfg(all(feature = "metal", not(feature = "cuda")))]
    {
        Device::new_metal(0).unwrap_or(Device::Cpu)
    }
    #[cfg(not(any(feature = "cuda", feature = "metal")))]
    {
        Device::Cpu
    }
}

pub fn sample_next_token(
    logits: &Tensor, 
    logits_processor: &mut LogitsProcessor
) -> Result<u32, candle_core::Error> {
    // Expected logits shape: [batch_size, seq_len, vocab_size]
    let seq_len = logits.dim(1)?;
    
    // Extract last sequence position logits using Candle's IndexOp: [batch=0, seq_len - 1, :]
    let next_token_logits = logits.i((0, seq_len - 1, ..))?;

    // Sample next token ID via temperature & top-p/top-k logic
    let next_token_id = logits_processor.sample(&next_token_logits)?;
    Ok(next_token_id)
}
